import {enrollWithCity,inviteForExchange,openDeviceSession,readDeviceFile,writeDeviceFile} from '../apps/client/device-enrollment.mjs';
import {copyFileSync,writeFileSync} from 'node:fs';
import {resolve} from 'node:path';
let input='';for await (const chunk of process.stdin)input+=chunk;
const auth=JSON.parse(input);input='';
const expected='031fdba6-e94c-4298-a095-6ff04a65481d';
if(auth.host!=='http://172.31.12.151:4391'||auth.cityId!==expected)throw new Error('EXPECTED_CITY_MISMATCH');
const h={Authorization:'Bearer '+auth.token,'Content-Type':'application/json','X-City-Api-Version':'0','X-City-Schema-Version':'0'};
const api=async(path,body)=>{const r=await fetch(auth.host+'/api/v0/'+path,{method:body?'POST':'GET',headers:h,body:body?JSON.stringify(body):undefined,signal:AbortSignal.timeout(10000)});const data=await r.json();if(!r.ok)throw Object.assign(new Error(data.detail||data.error||'HTTP_REFUSED'),{code:data.errorCode||r.status});return data;};
const city=await api('city');if(city.cityId!==expected)throw new Error('LIVE_CITY_MISMATCH');
const info=await api('pairing/info');if(info.activeSession||!['IDLE','USED','EXPIRED','LOCKED'].includes(info.sessionState))throw new Error('PAIRING_ACTIVE_PRESERVE_EXISTING_CODE');
const paired=await api('pairing/session',{expectedSessionState:info.sessionState});
const invite=inviteForExchange(paired.qrPayload);if(invite.cityId!==expected)throw new Error('PAIRING_CITY_MISMATCH');
const enrolled=await enrollWithCity({endpoint:auth.host,invite,displayName:'Alien · MERA-ALIANWARE',platform:'win32'});
await openDeviceSession(enrolled.record);
const file=resolve(process.env.LOCALAPPDATA,'Utopia/client/device-enrollment.json');
const old=readDeviceFile(file);if(old)copyFileSync(file,resolve('.runtime/private/retired-enrollment-before-mech.json'));
writeDeviceFile(file,enrolled.record);
const receipt={observedAt:new Date().toISOString(),sourceSha:'b06504f1f96984c960b2661b8ee3a7130796d379',endpoint:enrolled.record.endpoint,cityId:enrolled.record.cityId,deviceId:enrolled.record.deviceId,installationId:enrolled.record.installationId,displayName:enrolled.record.displayName,previousInstallationId:old?.installationId??null,previousInstallationState:'RETIRED',state:'FRESH_PAIRING_SESSION_VERIFIED'};
writeFileSync('.runtime/mech-member-pairing-receipt.json',JSON.stringify(receipt,null,2));console.log(JSON.stringify(receipt));
