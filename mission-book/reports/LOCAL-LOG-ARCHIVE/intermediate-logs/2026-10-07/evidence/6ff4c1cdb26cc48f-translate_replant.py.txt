def replant(stem,mapping):
    if (BASE/('replant/en/'+stem+'.md')).exists():return
    rel='replant/'+stem+'.md';src=(BASE/rel).read_text(encoding='utf-8').split('---',2)[2]
    out=src
    blocks=[]
    def protect(m):
        blocks.append(m[0]);return 'PRESERVED_CODE_BLOCK_'+str(len(blocks)-1)
    out=re.sub(r'```[^\n]*\n[\s\S]*?```',protect,out)
    # Long exact replacements before shorter labels.
    for a,b in sorted({**COMMON,**mapping}.items(),key=lambda x:len(x[0]),reverse=True):out=out.replace(a,b)
    out=out.replace('**Dependency Mission:** 无','**Dependency Mission:** none')
    remaining=[x for x in out.splitlines() if re.search('[\u4e00-\u9fff]',x)]
    if remaining:raise ValueError((stem,remaining))
    for i,block in enumerate(blocks):out=out.replace('PRESERVED_CODE_BLOCK_'+str(i),block)
    # All current source links stay targeted at the same resource after en/ relocation.
    out=re.sub(r'(\]\()([^\)]+)(\))',lambda m:m[0] if m[2].startswith(('http:','https:','#')) else m[1]+'../'+m[2]+m[3],out)
    save(rel,out,lang='en')
