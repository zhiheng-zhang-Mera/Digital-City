def patch_links(text,oldfile,newfile):
 # balanced Markdown destination parser supports district names with parentheses.
 out=[];pos=0
 while True:
  start=text.find('](',pos)
  if start<0:out.append(text[pos:]);break
  end=start+2;depth=1
  while end<len(text) and depth:
   if text[end]=='(':depth+=1
   elif text[end]==')':depth-=1
   end+=1
  if depth:out.append(text[pos:]);break
  value=text[start+2:end-1];newvalue=value
  if value and not re.match(r'[A-Za-z][\w+.-]*:',value) and not value.startswith('#'):
   bare=value.strip('<>');pathpart,sep,fragment=bare.partition('#'); decoded=urllib.parse.unquote(pathpart)
   target=os.path.normpath(os.path.join(os.path.dirname(oldfile),decoded)).replace('\\','/')
   newtarget=remap(target)
   if target!='.' and (newtarget!=target or newfile!=oldfile):
    newvalue=os.path.relpath(newtarget,os.path.dirname(newfile)).replace('\\','/').replace(' ','%20')+(sep+fragment if sep else '')
    if value.startswith('<'):newvalue='<'+newvalue+'>'
  out.append(text[pos:start+2]+newvalue+')');pos=end
 return ''.join(out)
