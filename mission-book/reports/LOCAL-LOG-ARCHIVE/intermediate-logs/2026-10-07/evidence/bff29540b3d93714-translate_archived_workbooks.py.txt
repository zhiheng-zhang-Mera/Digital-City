from pathlib import Path
import re, hashlib, json
ROOT=Path('\\\\?\\D:\\Digital-City-REX-20261006')
BASE=ROOT/'mission-book/finished'
def save(rel, body, lang='zh-CN'):
    source=BASE/rel
    before=source.read_bytes()
    text=before.decode('utf-8-sig')
    parts=re.split(r'^---\s*$',text,maxsplit=2,flags=re.M)
    assert len(parts)==3
    dest=source.parent/lang/source.name
    dest.parent.mkdir(exist_ok=True)
    header=('> 中文阅读译本 / Reading translation。此文件没有工作书元数据，也不赋予 authority。'
            '以下规则与状态属于历史归档；[canonical source](../'+source.name+') 的原始 frontmatter 是唯一元数据来源。\n\n')
    dest.write_text(header+body.strip()+'\n',encoding='utf-8')
    link='./'+lang+'/'+source.name
    current=source.read_bytes()
    assert current==before
    if (']('+link+')').encode() not in current:
        with source.open('ab') as f:f.write(('\n\n[阅读译本 / Reading translation]('+link+')\n').encode())
    assert source.read_bytes().startswith(before)
    assert re.split(r'^---\s*$',source.read_bytes().decode('utf-8-sig'),maxsplit=2,flags=re.M)[:2]==parts[:2]
    assert not re.match(r'^---',dest.read_text(encoding='utf-8'))
    for sha in re.findall(r'[0-9a-f]{40}',parts[2]):assert sha in body,(rel,sha)
    assert len(re.findall(r'^#{1,6} ',parts[2],re.M))==len(re.findall(r'^#{1,6} ',body,re.M)),rel
    print('VERIFIED',rel)
